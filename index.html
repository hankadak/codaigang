function doPost(e) { try { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("시트1"); var data = JSON.parse(e.postData.contents); sheet.appendRow([ data.timestamp || new Date().toLocaleString(), data.name, data.phone, data.address, data.region, data.items, data.subtotal, data.fee, data.delivery, data.totalPrice, data.receipt ]); return ContentService.createTextOutput(JSON.stringify({ "result": "success" })) .setMimeType(ContentService.MimeType.JSON); } catch(error) { return ContentService.createTextOutput(JSON.stringify({ "result": "error", "message": error.toString() })) .setMimeType(ContentService.MimeType.JSON); } }
